usage: jobhunt ingest [-h] [--mine DIR] [--course DIR] [--again] [--tidy] [folders ...]

positional arguments:
  folders       folders to sort; Hal proposes where each file goes

options:
  -h, --help    show this help message and exit
  --mine DIR    a folder of your own work, sorted too
  --course DIR  a folder of course material you were given, sorted too
  --again       read with the AI the documents an earlier ingest filed by rules alone
  --tidy        back up, then hold everything already in the record to today's rules
